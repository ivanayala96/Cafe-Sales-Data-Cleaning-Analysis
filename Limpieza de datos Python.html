df['Quantity'] = pd.to_numeric(df['Quantity'], errors='coerce').astype('float')
df['Price Per Unit'] = pd.to_numeric(df['Price Per Unit'], errors='coerce')
df['Total Spent'] = pd.to_numeric(df['Total Spent'],  errors='coerce')
df['Transaction Date'] = pd.to_datetime(df['Transaction Date'],format='%Y-%m-%d', errors='coerce')
mediana_fecha = df['Transaction Date'].dropna().median()

quantity = df['Quantity'].to_numpy()
price = df['Price Per Unit'].to_numpy()
total_spent = df['Total Spent'].to_numpy()

concepto = {
    'Item': 'Others',
    'Quantity': (df['Total Spent']/df['Price Per Unit']),
    'Price Per Unit': np.nanmean(price),
    'Total Spent': np.nanmean(total_spent),
    'Payment Method': 'Cash',
    'Location': 'In-Store',
    'Transaction Date': mediana_fecha
}

df['Transaction Date'] = df['Transaction Date'].fillna(mediana_fecha)
df = df.fillna(concepto)

basura = ['UNKNOWN', 'ERROR', 'Unknown', 'Error', 'unknown', 'error']

#Cambiamos la información en las columnas
df['Item'] = df['Item'].replace(basura, 'Others')
df['Quantity']= df['Quantity'].replace(basura, np.mean(quantity))
df['Price Per Unit']= df['Price Per Unit'].replace(basura, np.mean(price))
df['Total Spent']= df['Total Spent'].replace(basura, np.mean(total_spent))
df['Payment Method'] = df['Payment Method'].replace(basura, 'Cash')
df['Location'] = df['Location'].replace(basura, 'In-Store')
df['Transaction Date'] = df['Transaction Date'].replace(basura, mediana_fecha)
